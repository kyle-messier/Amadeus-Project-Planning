\subsection{Status Reporting}
\begin{planningbox}{Minimum status report content}
\begin{itemize}
  \item Overall status and reason: \StatusOnTrack\enspace \StatusAtRisk\enspace \StatusBlocked
  \item Outcomes and deliverables completed since the last report
  \item Next-period priorities and milestone forecast
  \item Schedule and budget variance against the approved baseline
  \item New or changed risks, issues, assumptions, and dependencies
  \item Decisions, support, or escalations required, each with an owner and due date
\end{itemize}
\end{planningbox}
